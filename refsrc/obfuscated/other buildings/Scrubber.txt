		v_u_11:Play()

local v_u_12 = game.ReplicatedStorage.Pollution

	if v_u_4:tickIsBusy() then
		return
	elseif v_u_8:canHoldXMoreOf("Residue", 4) then
		if v_u_6:hasPower(60000) then
			local v13 = v_u_7:getItem()
			if v_u_7:getBuffer() < 2 then
				v_u_9.Value = "idle"
			else
				local v_u_14
				if v13 == "Water" then
					v_u_14 = 1
				elseif v13 == "Filtered Water" then
					v_u_14 = 1.1
				elseif v13 == "Residue" then

						v_u_9.Value = "idle"
						return
					end
					v_u_14 = 1.939
				end
				v_u_9.Value = "run"
				v_u_7:removeItem(2)
				v_u_4:nextTicks(4, function()
					-- upvalues: (ref) v_u_6, (ref) v_u_8, (ref) v_u_12, (ref) v_u_14
					v_u_6:removePower(15000)
					v_u_8:addItem(1, "Residue")

					local v16 = v15.Value
					local v17 = v_u_14 * 450
					v15.Value = v16 - math.ceil(v17)
				end)
			end
		else
			v_u_9.Value = "idle"
			return
		end
	else
		v_u_9.Value = "idle"
		return
	end
end
