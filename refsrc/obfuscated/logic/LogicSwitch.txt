local v1 = game:GetService("ReplicatedStorage")
local v2 = require(v1.Modules.PowerUtilsV2)
require(v1.Modules.PipeUtilsV2)
require(v1.Modules.Enums.Items)
local v_u_3 = v2.loadConnector(script.Parent.MainConnector)
local v_u_4 = v2.loadConnector(script.Parent.MainConnector2)
local v5 = {
	["localUpdate"] = function() -- name: localUpdate end
}
local v_u_6 = script.Parent.ModelData.State
script.Parent.Switch.On.ProximityPrompt.Triggered:Connect(function()
	-- upvalues: (copy) v_u_6
	if v_u_6.Value == "1" then
		v_u_6.Value = "0"
		script.Parent.Switch.On.Valve:Play()
	elseif v_u_6.Value == "0" then
		v_u_6.Value = "1"
		script.Parent.Switch.On.Valve:Play()
	end
end)
function v5.tick() -- name: tick
	-- upvalues: (copy) v_u_6, (copy) v_u_3, (copy) v_u_4
	script.Parent.Switch.On.ProximityPrompt.Enabled = true
	if v_u_6.Value == "1" then
		script.Parent.Switch.On.Transparency = 0
		script.Parent.Switch.Off.Transparency = 1
		script.Parent.Switch.On.ProximityPrompt.ObjectText = "OPEN"
		if v_u_3:getBuffer() >= 1000 then
			v_u_4:addPower((v_u_3:removePower(1000)))
		end
	else
		if v_u_6.Value == "0" then
			script.Parent.Switch.On.Transparency = 1
			script.Parent.Switch.Off.Transparency = 0
			script.Parent.Switch.On.ProximityPrompt.ObjectText = "CLOSED"
			v_u_3:removePower(1000)
		end
		return
	end
end
return v5
